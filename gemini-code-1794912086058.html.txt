        <p class="subtitle">Teste palavras diferentes com a função <code>calcula_hash</code></p>

        <div class="input-group">
            <label for="palavra1">Palavra 1 (mínimo 2 letras):</label>
            <input type="text" id="palavra1" value="amor" oninput="processarHash()">
        </div>

        <div class="input-group">
            <label for="palavra2">Palavra 2 (mínimo 2 letras):</label>
            <input type="text" id="palavra2" value="roma" oninput="processarHash()">
        </div>

        <div class="results">
            <div class="hash-display">
                <span>Hash P1:</span>
                <span id="valorHash1">0</span>
            </div>
            <div class="hash-display">
                <span>Hash P2:</span>
                <span id="valorHash2">0</span>
            </div>
            <div id="statusMensagem" class="status match">Colisão Encontrada!</div>
        </div>

        <div class="preset-buttons">
            <button onclick="carregarExemplo('amor', 'roma')">Amor / Roma</button>
            <button onclick="carregarExemplo('alto', 'lato')">Alto / Lato</button>
            <button onclick="carregarExemplo('ar', 'ra')">Ar / Ra</button>

    <script>
        // Função de hash simples baseada na soma dos códigos ASCII dos caracteres
        function calcula_hash(palavra) {
            let hash = 0;
            for (let i = 0; i < palavra.length; i++) {
                hash += palavra.charCodeAt(i);
            }
            return hash;
        }

        function processarHash() {
            const p1 = document.getElementById('palavra1').value;
            const p2 = document.getElementById('palavra2').value;

            const h1 = calcula_hash(p1);
            const h2 = calcula_hash(p2);

            document.getElementById('valorHash1').innerText = h1;
            document.getElementById('valorHash2').innerText = h2;

            const statusEl = document.getElementById('statusMensagem');

            // Verifica se são diferentes e possuem o mesmo hash
            if (p1 !== p2 && p1.length >= 2 && p2.length >= 2 && h1 === h2) {
                statusEl.className = "status match";
                statusEl.innerText = "✨ Colisão Encontrada! Mesmo hash para palavras diferentes.";
            } else if (p1 === p2) {
                statusEl.className = "status no-match";
                statusEl.innerText = "⚠️ Digite palavras diferentes.";
            } else {
                statusEl.className = "status no-match";
                statusEl.innerText = "❌ Sem colisão (Hashes diferentes).";
            }
        }

        function carregarExemplo(p1, p2) {
            document.getElementById('palavra1').value = p1;
            document.getElementById('palavra2').value = p2;
            processarHash();
        }

        // Executa no carregamento inicial
        processarHash();
    </script>
